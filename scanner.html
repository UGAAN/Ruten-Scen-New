Setup Guide:
  1. Create or open a sheet in your Google Drive.
  2. Click Extensions > Apps Script in the top menu.
  3. Delete any default code and paste the Apps Script code below: 
    function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({ result: "success", status: "active", message: "Google Sheet Return Sync is ready!" })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) throw new Error("Spreadsheet not found. Open this in Extensions > Apps Script inside your sheet!");
    var sheet = ss.getActiveSheet();
    var defaultHeaders = ["Date", "Scan Time", "PC Name", "OMS", "Return AWB Number", "Master SKU", "Marketplace SKU", "Marketplace", "Suborder Number", "Return Quantity", "Condition", "Status", "Image URL"];

    if (sheet.getLastRow() === 0) {
      sheet.appendRow(defaultHeaders);
      sheet.getRange(1, 1, 1, defaultHeaders.length).setFontWeight("bold").setBackground("#dcfce7");
    }

    var contents = (e && e.postData && e.postData.contents) ? e.postData.contents : "";
    if (!contents) throw new Error("No data in POST request");
    var data = JSON.parse(contents);
    var items = Array.isArray(data) ? data : [data];
    if (items.length === 0) return ContentService.createTextOutput(JSON.stringify({ result: "success", count: 0 })).setMimeType(ContentService.MimeType.JSON);

    var lastCol = Math.max(sheet.getLastColumn(), 1);
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];

    // If existing sheet has no PC Name or OMS column, auto-add to header
    var pcColFound = false;
    var omsColFound = false;
    for (var h = 0; h < headers.length; h++) {
      var headerStr = String(headers[h]).toLowerCase();
      if (headerStr.includes("pc") || headerStr.includes("station")) pcColFound = true;
      if (headerStr === "oms" || headerStr.includes("source")) omsColFound = true;
    }
    if (!pcColFound && sheet.getLastRow() > 0) {
      sheet.getRange(1, lastCol + 1).setValue("PC Name").setFontWeight("bold").setBackground("#dcfce7");
      headers.push("PC Name");
      lastCol = lastCol + 1;
    }
    if (!omsColFound && sheet.getLastRow() > 0) {
      sheet.getRange(1, lastCol + 1).setValue("OMS").setFontWeight("bold").setBackground("#dcfce7");
      headers.push("OMS");
      lastCol = lastCol + 1;
    }

    var rows = items.map(function(item) {
      return headers.map(function(col) {
        var c = String(col).trim().toLowerCase();
        if (c.includes("date")) return item.date || "";
        if (c.includes("time")) return item.timestamp || "";
        if (c.includes("pc") || c.includes("station")) return item.pc_name || "PC-1";
        if (c === "oms" || c.includes("source")) return item.oms_source || "V2";
        if (c.includes("awb")) return item.awb || "";
        if (c.includes("master")) return item.master_sku || "";
        if (c.includes("market") && c.includes("sku")) return item.marketplace_sku || "";
        if (c.includes("suborder")) return item.suborder_no || "";
        if (c.includes("market")) return item.marketplace || "";
        if (c.includes("qty") || c.includes("quantity")) return item.quantity || 1;
        if (c.includes("condition")) return item.condition || "Good";
        if (c.includes("status")) return item.status || "Submitted";
        if (c.includes("image")) return item.image_url || "";
        return "";
      });
    });

    if (rows.length > 0) {
      sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, headers.length).setValues(rows);
    }
    return ContentService.createTextOutput(JSON.stringify({ result: "success", count: rows.length })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
